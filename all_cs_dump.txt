    67	    public IReadOnlyList<CspEnumeratedPrinter> GetPrinterPortPairs(int printerFilter)
    68	    {
    69	        const int bufferSize = 512;
    70	        _ = InitializePrinter(printerFilter);
    71	
    72	        var buffer = Marshal.AllocHGlobal(bufferSize);
    73	        try
    74	        {
    75	            Span<byte> cleared = stackalloc byte[bufferSize];
    76	            Marshal.Copy(cleared.ToArray(), 0, buffer, bufferSize);
    77	
    78	            var count = _getPrinterPortNum(buffer, bufferSize);
    79	            if (count <= 0)
    80	            {
    81	                return Array.Empty<CspEnumeratedPrinter>();
    82	            }
    83	
    84	            var raw = new byte[bufferSize];
    85	            Marshal.Copy(buffer, raw, 0, bufferSize);
    86	
    87	            var results = new List<CspEnumeratedPrinter>();
    88	            var maxPairs = Math.Min(count, bufferSize / 2);
    89	            for (var i = 0; i < maxPairs; i++)
    90	            {
    91	                var offset = i * 2;
    92	                var modelCode = raw[offset];
    93	                var deviceIndex = raw[offset + 1];
    94	                if (modelCode == 0 && deviceIndex == 0)
    95	                {
    96	                    break;
    97	                }

   100	            }

   138	    private static nint LoadLibrary(string? explicitPath, out string loadedPath)
   139	    {
   140	        var candidates = BuildCandidates(explicitPath).Distinct(StringComparer.OrdinalIgnoreCase).ToArray();
   141	        var failures = new List<string>();
   142	
   143	        foreach (var candidate in candidates)
   144	        {
   145	            try
   146	            {
   147	                if (NativeLibrary.TryLoad(candidate, out var handle) && handle != nint.Zero)
   148	                {
   149	                    loadedPath = candidate;
   150	                    return handle;
   151	                }
   152	            }
   153	            catch (BadImageFormatException ex)
   154	            {
   155	                failures.Add($"{candidate}: wrong DLL architecture for this {PreferredArchitectureLabel} process ({ex.Message})");
   156	            }
   157	            catch (Exception ex)
   158	            {
   159	                failures.Add($"{candidate}: {ex.Message}");
   160	            }
   161	        }
   162	
   163	        throw new DllNotFoundException(
   164	            "Could not load cspstat library. Tried: " + string.Join(", ", candidates) + ". " +
   165	            $@"This executable is running as {PreferredArchitectureLabel} and expects {PreferredDllName}, ideally in .\dll\{PreferredDllName}. " +
   166	            $"You can also pass --dll-path <absolute-path-to-{PreferredDllName}>." +
   167	            (failures.Count == 0 ? string.Empty : " Failures: " + string.Join(" | ", failures)));
   168	    }

   170	    private static IEnumerable<string> BuildCandidates(string? explicitPath)
   171	    {
   172	        foreach (var candidate in ExpandConfiguredPath(explicitPath))
   173	        {
   174	            yield return candidate;
   175	        }
   176	
   177	        foreach (var candidate in ExpandConfiguredPath(Environment.GetEnvironmentVariable("DNP_CSPSTAT_PATH")))
   178	        {
   179	            yield return candidate;
   180	        }
   181	
   182	        var baseDir = AppContext.BaseDirectory;
   183	        var preferredNames = new[]
   184	        {
   185	            Path.Combine("dll", PreferredDllName),
   186	            Path.Combine("dll", PreferredArchitectureLabel == "x64" ? "win-x64" : "win-x86", PreferredDllName),
   187	            PreferredDllName,
   188	            Path.Combine(PreferredArchitectureLabel == "x64" ? @"Native\win-x64" : @"Native\win-x86", PreferredDllName),
   189	            Path.Combine("dll", FallbackDllName),
   190	            FallbackDllName,
   191	            Path.Combine(PreferredArchitectureLabel == "x64" ? @"Native\win-x86" : @"Native\win-x64", FallbackDllName)
   192	        };
   193	
   194	        foreach (var name in preferredNames)
   195	        {
   196	            yield return Path.Combine(baseDir, name);
   197	        }
   198	
   199	        foreach (var name in preferredNames)
   200	        {
   201	            yield return name;
   202	        }
   203	    }

   205	    private static IEnumerable<string> ExpandConfiguredPath(string? configuredPath)
   206	    {
   207	        if (string.IsNullOrWhiteSpace(configuredPath))
   208	        {
   209	            yield break;
   210	        }
   211	
   212	        var trimmed = configuredPath.Trim();
   213	        if (Directory.Exists(trimmed))
   214	        {
   215	            yield return Path.Combine(trimmed, PreferredDllName);
   216	            yield return Path.Combine(trimmed, FallbackDllName);
   217	            yield break;
   218	        }
   219	
   220	        yield return trimmed;
   221	    }

    43	               || DnpModelResolver.IsPotentialDnpPrinterText(Manufacturer)

    47	    public bool MatchesSelector(LinuxUsbDeviceSelector selector)
    48	    {
    49	        ArgumentNullException.ThrowIfNull(selector);
    50	
    51	        if (!string.IsNullOrWhiteSpace(selector.DevicePath)
    52	            && !string.Equals(DevicePath, selector.DevicePath, StringComparison.OrdinalIgnoreCase))
    53	        {
    54	            return false;
    55	        }
    56	
    57	        if (!string.IsNullOrWhiteSpace(selector.VendorId)
    58	            && !string.Equals(VendorId, selector.VendorId, StringComparison.OrdinalIgnoreCase))
    59	        {
    60	            return false;
    61	        }
    62	
    63	        if (!string.IsNullOrWhiteSpace(selector.ProductId)
    64	            && !string.Equals(ProductId, selector.ProductId, StringComparison.OrdinalIgnoreCase))
    65	        {
    66	            return false;
    67	        }
    68	
    69	        if (!string.IsNullOrWhiteSpace(selector.SerialNumber)
    70	            && !string.Equals(SerialNumber, selector.SerialNumber, StringComparison.Ordinal))
    71	        {
    72	            return false;
    73	        }
    74	
    75	        if (selector.BusNumber.HasValue && BusNumber != selector.BusNumber)
    76	        {
    77	            return false;
    78	        }

    89	public sealed record LinuxUsbDeviceSelector(
    90	    string? DevicePath,
    91	    string? VendorId,
    92	    string? ProductId,
    93	    string? SerialNumber,
    94	    int? BusNumber,
    95	    int? DeviceAddress)
    96	{
    97	    public static bool TryParse(string? text, out LinuxUsbDeviceSelector? selector)
    98	    {
    99	        selector = null;
   100	        if (string.IsNullOrWhiteSpace(text))
   101	        {
   102	            return false;
   103	        }
   104	
   105	        var trimmed = text.Trim();
   106	        if (!trimmed.StartsWith("usb:", StringComparison.OrdinalIgnoreCase))
   107	        {
   108	            return false;
   109	        }
   110	
   111	        string? devicePath = null;
   112	        string? vendorId = null;
   113	        string? productId = null;
   114	        string? serialNumber = null;
   115	        int? busNumber = null;
   116	        int? deviceAddress = null;
   117	
   118	        foreach (var part in trimmed[4..].Split(',', StringSplitOptions.RemoveEmptyEntries | StringSplitOptions.TrimEntries))
   119	        {
   120	            var index = part.IndexOf('=');
   121	            if (index <= 0 || index >= part.Length - 1)
   122	            {
   123	                continue;
   124	            }
   125	
   126	            var key = part[..index].Trim().ToLowerInvariant();
   127	            var value = part[(index + 1)..].Trim();
   128	            if (value.Length == 0)
   129	            {
   130	                continue;
   131	            }
   132	
   133	            switch (key)
   134	            {
   135	                case "path":
   136	                    devicePath = value;
   137	                    break;
   138	                case "vid":
   139	                    vendorId = NormalizeHex(value);
   140	                    break;
   141	                case "pid":
   142	                    productId = NormalizeHex(value);
   143	                    break;
   144	                case "serial":
   145	                    serialNumber = value;
   146	                    break;
   147	                case "bus":
   148	                    busNumber = ParseInt(value);
   149	                    break;
   150	                case "addr":
   151	                case "device":
   152	                case "dev":
   153	                    deviceAddress = ParseInt(value);
   154	                    break;
   155	            }
   156	        }
   157	
   158	        selector = new LinuxUsbDeviceSelector(devicePath, vendorId, productId, serialNumber, busNumber, deviceAddress);
   159	        return true;
   160	    }
   161	
   162	    private static int? ParseInt(string text)
   163	        => int.TryParse(text, NumberStyles.Integer, CultureInfo.InvariantCulture, out var value) ? value : null;
   164	
   165	    private static string NormalizeHex(string value)
   166	        => value.Trim().TrimStart('0').Length == 0
   167	            ? "0000"
   168	            : value.Trim().TrimStart('0').PadLeft(4, '0').ToLowerInvariant();
   169	}

   172	{
   173	    private static readonly string[] DefaultDeviceCandidates =
   174	    [
   175	        "/dev/usb/lp0",
   176	        "/dev/usb/lp1",
   177	        "/dev/usb/lp2",
   178	        "/dev/usb/lp3",
   179	        "/dev/lp0",
   180	        "/dev/lp1",
   181	        "/dev/lp2"
   182	    ];

   184	    public static IReadOnlyList<LinuxUsbDeviceInfo> Enumerate(string? preferredDevicePath = null)
   185	    {
   186	        var results = new List<LinuxUsbDeviceInfo>();
   187	        var seen = new HashSet<string>(StringComparer.OrdinalIgnoreCase);
   188	
   189	        foreach (var path in GetDeviceCandidates(preferredDevicePath))
   190	        {
   191	            if (!seen.Add(path))
   192	            {
   193	                continue;
   194	            }
   195	
   196	            results.Add(ReadUsbDeviceInfo(path));
   197	        }
   198	
   199	        return results;
   200	    }
   201	
   202	    public static LinuxUsbDeviceInfo? Resolve(string? queryValue)
   203	    {
   204	        if (string.IsNullOrWhiteSpace(queryValue))
   205	        {
   206	            return Enumerate().FirstOrDefault();
   207	        }
   208	
   209	        var trimmed = queryValue.Trim();
   210	        if (!LinuxUsbDeviceSelector.TryParse(trimmed, out var selector) || selector is null)
   211	        {
   212	            return File.Exists(trimmed) ? ReadUsbDeviceInfo(trimmed) : null;
   213	        }
   214	
   215	        return Enumerate(selector.DevicePath)
   216	            .FirstOrDefault(device => device.MatchesSelector(selector));
   217	    }
   218	
   219	    private static IEnumerable<string> GetDeviceCandidates(string? preferredDevicePath)
   220	    {
   221	        if (!string.IsNullOrWhiteSpace(preferredDevicePath) && !preferredDevicePath.Trim().StartsWith("usb:", StringComparison.OrdinalIgnoreCase))
   222	        {
   223	            yield return preferredDevicePath.Trim();
   224	        }
   225	
   226	        foreach (var path in DefaultDeviceCandidates)
   227	        {
   228	            if (File.Exists(path))
   229	            {
   230	                yield return path;
   231	            }
   232	        }
   233	    }

   235	    private static LinuxUsbDeviceInfo ReadUsbDeviceInfo(string devicePath)
   236	    {
   237	        var baseName = Path.GetFileName(devicePath);
   238	        var candidateRoots = new[]
   239	        {
   240	            Path.Combine("/sys/class/usb", baseName),
   241	            Path.Combine("/sys/class/usblp", baseName)
   242	        };
   243	
   244	        foreach (var root in candidateRoots)
   245	        {
   246	            if (!Directory.Exists(root))
   247	            {
   248	                continue;
   249	            }
   250	
   251	            var metadata = ReadMetadataFromAncestors(Path.Combine(root, "device"));
   252	            return new LinuxUsbDeviceInfo(
   253	                DevicePath: devicePath,
   254	                ProductName: metadata.ProductName,
   255	                Manufacturer: metadata.Manufacturer,
   256	                VendorId: metadata.VendorId,
   257	                ProductId: metadata.ProductId,
   258	                SerialNumber: metadata.SerialNumber,
   259	                BusNumber: metadata.BusNumber,
   260	                DeviceAddress: metadata.DeviceAddress);

    24	    public async Task<byte[]> QueryAsync(DnpCommand command, CancellationToken cancellationToken = default)
    25	    {
    26	        ArgumentNullException.ThrowIfNull(command);
    27	
    28	        if (!OperatingSystem.IsLinux())
    29	        {
    30	            throw new PlatformNotSupportedException("Linux transport can only run on Linux.");
    31	        }
    32	
    33	        var devicePath = ResolveDevicePath();
    34	        var fileOptions = new FileStreamOptions
    35	        {
    36	            Access = FileAccess.ReadWrite,
    37	            Mode = FileMode.Open,
    38	            Share = FileShare.ReadWrite,
    39	            BufferSize = _readChunkSize,
    40	            Options = FileOptions.Asynchronous | FileOptions.WriteThrough
    41	        };

    43	        await using var stream = new FileStream(devicePath, fileOptions);

    48	
    49	        if (_postWriteDelayMs > 0)
    50	        {
    51	            await Task.Delay(_postWriteDelayMs, cancellationToken).ConfigureAwait(false);
    52	        }
    53

    57	    private async Task<byte[]> ReadResponseAsync(FileStream stream, CancellationToken cancellationToken)
    58	    {
    59	        var buffer = new byte[_readChunkSize];
    60	        using var response = new MemoryStream();
    61	
    62	        while (true)
    63	        {
    64	            cancellationToken.ThrowIfCancellationRequested();
    65	
    66	            var bytesRead = await stream.ReadAsync(buffer, cancellationToken)
    67	                .AsTask()
    68	                .WaitAsync(TimeSpan.FromMilliseconds(_readTimeoutMs), cancellationToken)
    69	                .ConfigureAwait(false);
    70	
    71	            if (bytesRead <= 0)
    72	            {
    73	                if (response.Length == 0)
    74	                {
    75	                    throw new TimeoutException($"No response received from device within {_readTimeoutMs} ms.");
    76	                }
    77	
    78	                break;
    79	            }
    80	
    81	            response.Write(buffer, 0, bytesRead);
    82	
    83	            if (DnpPacketCodec.TryGetExpectedResponseLength(response.GetBuffer().AsSpan(0, checked((int)response.Length)), out var expectedLength)
    84	                && response.Length >= expectedLength)
    85	            {
    86	                break;
    87	            }
    88	        }
    89	
    90	        return response.ToArray();
    91	    }

    53	    public static byte[] Encode(DnpCommand command)
    54	    {
    55	        ArgumentNullException.ThrowIfNull(command);
    56	
    57	        var payload = command.Payload ?? Array.Empty<byte>();
    58	        var arg1 = PadAscii(command.Arg1, 6);
    59	        var arg2 = PadAscii(command.Arg2, 16);

    72	    public static string DecodeAsciiPayload(byte[] responseBytes)
    73	    {
    74	        ArgumentNullException.ThrowIfNull(responseBytes);
    75	
    76	        if (TryGetExpectedResponseLength(responseBytes, out var expectedLength) && responseBytes.Length >= expectedLength)
    77	        {
    78	            var payloadLengthText = Encoding.ASCII.GetString(responseBytes, 0, 8);
    79	            if (int.TryParse(payloadLengthText, out var payloadLength) && responseBytes.Length >= 8 + payloadLength)
    80	            {
    81	                return Encoding.ASCII.GetString(responseBytes, 8, payloadLength).TrimEnd('\0', '\r', '\n');
    82	            }
    83	        }
    84	
    85	        return Encoding.ASCII.GetString(responseBytes).TrimEnd('\0', '\r', '\n');
    86	    }

    93	            if (int.TryParse(payloadLengthText, out var payloadLength))
    94	            {
    95	                expectedLength = 8 + payloadLength;
    96	                return true;
    97	            }
    98	        }
    99	
   100	        expectedLength = 0;
   101	        return false;

    59	    private static bool TryParseCspStatus(string raw, out PrinterStatusInfo info)
    60	    {
    61	        var normalized = raw.Trim();
    62	        if (normalized.StartsWith("CSP:", StringComparison.OrdinalIgnoreCase))
    63	        {
    64	            normalized = normalized[4..];
    65	        }
    66	
    67	        int status;
    68	        if (normalized.StartsWith("0x", StringComparison.OrdinalIgnoreCase))
    69	        {
    70	            if (!int.TryParse(normalized[2..], System.Globalization.NumberStyles.HexNumber, System.Globalization.CultureInfo.InvariantCulture, out status))
    71	            {
    72	                info = new PrinterStatusInfo(raw, PrinterStatusKind.Unknown, "Unknown");
    73	                return false;
    74	            }
    75	        }
    76	        else if (!int.TryParse(normalized, out status))
    77	        {
    78	            info = new PrinterStatusInfo(raw, PrinterStatusKind.Unknown, "Unknown");
    79	            return false;
    80	        }
    81	
    82	        info = status switch
    83	        {
    84	            0x00010001 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.Idle, "Idle"),
    85	            0x00010002 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.Printing, "Printing"),
    86	            0x00010020 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.Cooling, "Cooling"),
    87	            0x00020001 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.CoverOpen, "Cover open"),
    88	            0x00010008 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.PaperEnd, "Paper end"),
    89	            0x00010010 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.RibbonEnd, "Ribbon end"),
    90	            0x00020002 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.PaperJam, "Paper jam"),
    91	            0x00020004 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.RibbonError, "Ribbon error"),
    92	            0x00020008 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.PaperDefinitionError, "Paper error"),
    93	            0x00020020 => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.SystemError, "Scrapbox error"),
    94	            _ => new PrinterStatusInfo($"0x{status:X8}", PrinterStatusKind.Unknown, "Unknown")
    95	        };
    96	        return true;
    97	    }
    98	
    99	    private static int? ParseIntegerAfterPrefix(string raw, int prefixLength)
   100	    {
   101	        if (raw.Length <= prefixLength)
   102	        {
   103	            return ExtractTrailingInt(raw);
   104	        }
   105	
   106	        return int.TryParse(new string(raw[prefixLength..].Where(char.IsDigit).ToArray()), out var value)
   107	            ? value
   108	            : ExtractTrailingInt(raw);
   109	    }
   110	
   111	    private static int? ExtractTrailingInt(string raw)
   112	    {
   113	        var match = TrailingNumberRegex().Match(raw);
   114	        return match.Success && int.TryParse(match.Value, out var value)
   115	            ? value
   116	            : null;
   117	    }
   118	
   119	    private static string? ExtractFirstDigits(string raw, int minimumLength)
   120	    {
   121	        var match = DigitRunRegex().Match(raw);
   122	        return match.Success && match.Value.Length >= minimumLength
   123	            ? match.Value
   124	            : null;
   125	    }
   126	
   127	    [GeneratedRegex(@"\d+")]
   128	    private static partial Regex DigitRunRegex();
   129	
   130	    [GeneratedRegex(@"\d+$")]
   131	    private static partial Regex TrailingNumberRegex();
   132	}
